CREATE TABLE default.XDR_TRACE_ALERT
(
    `snow_id` String COMMENT '告警id',
    `task_id` String COMMENT '溯源任务id',
    `execute_record_id` String COMMENT '执行记录id',
    `alert_time` DateTime COMMENT '发生时间',
    `alert_name` String COMMENT '告警名称',
    `alert_desc` String COMMENT '告警描述',
    `affected_ip` String COMMENT '受影响IP',
    `src_ip` String COMMENT '源IP',
    `dst_ip` String COMMENT '目的IP',
    `protocol` LowCardinality(String) COMMENT '协议',
    `file_direction` LowCardinality(String) COMMENT '文件传输方向',

    `insert_time` DateTime DEFAULT now() COMMENT '插入时间'
)
